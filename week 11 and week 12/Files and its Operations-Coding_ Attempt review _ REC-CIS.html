import string
import sys
def reverse_line():
input_data = sys.stdin.read().splitlines()
if not input_data:
return
filename = input_data[0].strip()
line_number = int(input_data[1].strip())
with open(filename, "r") as f:
lines = f.readlines()
target_idx = line_number - 1
if 0 <= target_idx < len(lines):
line = lines[target_idx]
# 1. Separate newline
if line.endswith("\r\n"):
ending = "\r\n"
text = line[:-2]
elif line.endswith("\n"):
ending = "\n"
text = line[:-1]
else:
ending = ""
text = line
# 2. Separate trailing punctuation (e.g., '.', '!', '?')
punct = ""
while text and text[-1] in string.punctuation:
punct = text[-1] + punct
text = text[:-1]
# 3. Reverse only the word text and reattach punctuation + newline
lines[target_idx] = text[::-1] + punct + ending
with open("output.txt", "w") as f:
f.writelines(lines)
if __name__ == "__main__":
reverse_line()

def insert_line_at_position():

if not input_data:
return

new_line_content = input_data[2] # Content to insert
with open(input_filename, "r") as f:
lines = f.readlines()
insert_idx = line_number - 1
# Ensure existing line before insertion point has a trailing newline
if insert_idx > 0 and insert_idx <= len(lines):
prev_idx = insert_idx - 1
if not lines[prev_idx].endswith("\n"):
lines[prev_idx] += "\n"
elif insert_idx > len(lines) and lines and not lines[-1].endswith("\n"):
lines[-1] += "\n"
# Ensure new line content ends with a newline
if not new_line_content.endswith("\n"):
new_line_content += "\n"
# Insert new line at position or append at the end
if insert_idx >= len(lines):
lines.append(new_line_content)
else:
lines.insert(insert_idx, new_line_content)
with open("output.txt", "w") as f:
f.writelines(lines)
if __name__ == "__main__":
insert_line_at_position()

import string
import sys
def count_word_frequency():
input_text = sys.stdin.read().strip()
if not input_text:
return
# Split into raw tokens
raw_words = input_text.split()
# Clean words: remove punctuation and convert to lowercase
freq_map = {}
for word in raw_words:
cleaned_word = word.strip(string.punctuation).lower()
if cleaned_word:
freq_map[cleaned_word] = freq_map.get(cleaned_word, 0) + 1
# Sort words alphabetically
sorted_words = sorted(freq_map.keys())
# Format output as "word: count"
output_lines = [f"{word}: {freq_map[word]}\n" for word in sorted_words]
# Write to output.txt
with open("output.txt", "w") as f:
f.writelines(output_lines)
if __name__ == "__main__":
count_word_frequency()

import sys
def find_palindromes():
# Read the input filename from standard input
input_data = sys.stdin.read().splitlines()
if not input_data:
return
filename = input_data[0].strip()
with open(filename, "r") as f:

# Split into raw words
words = content.split()
palindromes = []
for word in words:
# Strip trailing/leading punctuation
cleaned = word.strip(string.punctuation)
# Check if the word is a valid palindrome (and longer than 1 character)

cleaned
and len(cleaned) > 1
and cleaned.lower() == cleaned.lower()[::-1]
):
palindromes.append(cleaned)
